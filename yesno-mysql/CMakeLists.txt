DISABLE_MISSING_PROFILE_WARNING()
ADD_DEFINITIONS(-DMYSQL_SERVER)

GET_FILENAME_COMPONENT(YESNO_PLUGIN_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}"
                       REALPATH)
SET(YESNO_C_PREBUILT_LIBRARY "" CACHE FILEPATH
    "Prebuilt libyesno_c.a supplied by an outer build system")
SET(YESNO_C_HEADER "" CACHE FILEPATH
    "yesno.h paired with YESNO_C_PREBUILT_LIBRARY")
OPTION(YESNO_WITH_FLIGHT "Build the remote Arrow Flight backend" OFF)
SET(YESNO_FLIGHT_CPP_PREBUILT_LIBRARY "" CACHE FILEPATH
    "Prebuilt libyesno_flight_cpp.a supplied by an outer build system")
SET(YESNO_FLIGHT_CPP_HEADER "" CACHE FILEPATH
    "client.h paired with YESNO_FLIGHT_CPP_PREBUILT_LIBRARY")
IF(EXISTS "${YESNO_PLUGIN_SOURCE_DIR}/flight-cpp/CMakeLists.txt")
  SET(YESNO_FLIGHT_CPP_DEFAULT_SOURCE_DIR
      "${YESNO_PLUGIN_SOURCE_DIR}/flight-cpp")
ELSE()
  SET(YESNO_FLIGHT_CPP_DEFAULT_SOURCE_DIR
      "${YESNO_PLUGIN_SOURCE_DIR}/../yesno-flight-c++")
ENDIF()
SET(YESNO_FLIGHT_CPP_SOURCE_DIR
    "${YESNO_FLIGHT_CPP_DEFAULT_SOURCE_DIR}" CACHE PATH
    "Source directory for the native yesno Flight C++ client")

IF(YESNO_C_PREBUILT_LIBRARY)
  IF(NOT YESNO_C_HEADER)
    MESSAGE(FATAL_ERROR
      "YESNO_C_HEADER is required with YESNO_C_PREBUILT_LIBRARY")
  ENDIF()
  SET(YESNO_C_LIBRARY "${YESNO_C_PREBUILT_LIBRARY}")
  GET_FILENAME_COMPONENT(YESNO_C_INCLUDE_DIR "${YESNO_C_HEADER}" DIRECTORY)
ELSE()
  FIND_PROGRAM(CARGO_EXECUTABLE cargo REQUIRED)
  SET(YESNO_C_SOURCE_DIR "${YESNO_PLUGIN_SOURCE_DIR}/../yesno-c")
  SET(YESNO_CORE_SOURCE_DIR "${YESNO_PLUGIN_SOURCE_DIR}/../yesno-core")
  SET(YESNO_C_MANIFEST "${YESNO_C_SOURCE_DIR}/Cargo.toml")
  SET(YESNO_C_TARGET_DIR "${CMAKE_CURRENT_BINARY_DIR}/rust-target")
  SET(YESNO_C_LIBRARY "${YESNO_C_TARGET_DIR}/release/libyesno_c.a")
  SET(YESNO_C_INCLUDE_DIR "${YESNO_C_SOURCE_DIR}/include")

  FILE(GLOB_RECURSE YESNO_C_SOURCES CONFIGURE_DEPENDS
       "${YESNO_C_SOURCE_DIR}/src/*.rs"
       "${YESNO_CORE_SOURCE_DIR}/src/*.rs")

  ADD_CUSTOM_COMMAND(
    OUTPUT "${YESNO_C_LIBRARY}"
    COMMAND "${CMAKE_COMMAND}" -E env
            "CARGO_TARGET_DIR=${YESNO_C_TARGET_DIR}"
            "RUSTFLAGS=-C relocation-model=pic"
            "${CARGO_EXECUTABLE}" build --release --locked
            --manifest-path "${YESNO_C_MANIFEST}"
    DEPENDS
      "${YESNO_C_MANIFEST}"
      "${YESNO_C_SOURCE_DIR}/Cargo.lock"
      "${YESNO_CORE_SOURCE_DIR}/Cargo.toml"
      ${YESNO_C_SOURCES}
    WORKING_DIRECTORY "${YESNO_C_SOURCE_DIR}"
    COMMENT "Building the standalone yesno C library"
    VERBATIM)

  ADD_CUSTOM_TARGET(yesno_c_build DEPENDS "${YESNO_C_LIBRARY}")
ENDIF()

ADD_LIBRARY(yesno_c STATIC IMPORTED GLOBAL)
SET_TARGET_PROPERTIES(yesno_c PROPERTIES
                      IMPORTED_LOCATION "${YESNO_C_LIBRARY}")

SET(YESNO_PLUGIN_SOURCES ha_yesno.cc backend_embedded.cc)
SET(YESNO_PLUGIN_LIBRARIES yesno_c extra::rapidjson ${CMAKE_DL_LIBS})

IF(YESNO_WITH_FLIGHT)
  FIND_PACKAGE(Arrow CONFIG REQUIRED)
  FIND_PACKAGE(ArrowFlight CONFIG REQUIRED)
  IF(YESNO_FLIGHT_CPP_PREBUILT_LIBRARY)
    IF(NOT YESNO_FLIGHT_CPP_HEADER)
      MESSAGE(FATAL_ERROR
        "YESNO_FLIGHT_CPP_HEADER is required with "
        "YESNO_FLIGHT_CPP_PREBUILT_LIBRARY")
    ENDIF()
    ADD_LIBRARY(yesno_flight_cpp STATIC IMPORTED GLOBAL)
    SET_TARGET_PROPERTIES(
      yesno_flight_cpp PROPERTIES
      IMPORTED_LOCATION "${YESNO_FLIGHT_CPP_PREBUILT_LIBRARY}")
    GET_FILENAME_COMPONENT(YESNO_FLIGHT_CPP_INCLUDE_DIR
                           "${YESNO_FLIGHT_CPP_HEADER}" DIRECTORY)
    TARGET_INCLUDE_DIRECTORIES(
      yesno_flight_cpp INTERFACE "${YESNO_FLIGHT_CPP_INCLUDE_DIR}/../../..")
    IF(TARGET ArrowFlight::arrow_flight_shared)
      TARGET_LINK_LIBRARIES(
        yesno_flight_cpp INTERFACE ArrowFlight::arrow_flight_shared)
    ELSEIF(TARGET ArrowFlight::arrow_flight_static)
      TARGET_LINK_LIBRARIES(
        yesno_flight_cpp INTERFACE ArrowFlight::arrow_flight_static)
    ELSE()
      MESSAGE(FATAL_ERROR
        "ArrowFlight package exports no supported Flight target")
    ENDIF()
  ELSE()
    SET(YESNO_FLIGHT_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    ADD_SUBDIRECTORY("${YESNO_FLIGHT_CPP_SOURCE_DIR}"
                     "${CMAKE_CURRENT_BINARY_DIR}/yesno-flight-cpp")
  ENDIF()
  LIST(APPEND YESNO_PLUGIN_SOURCES backend_flight.cc)
  LIST(APPEND YESNO_PLUGIN_LIBRARIES yesno_flight_cpp)
ENDIF()

IF(WITH_YESNO_STORAGE_ENGINE AND NOT WITHOUT_YESNO_STORAGE_ENGINE)
  MYSQL_ADD_PLUGIN(yesno ${YESNO_PLUGIN_SOURCES}
    STORAGE_ENGINE
    DEFAULT
    LINK_LIBRARIES ${YESNO_PLUGIN_LIBRARIES}
    )
ELSEIF(NOT WITHOUT_YESNO_STORAGE_ENGINE)
  MYSQL_ADD_PLUGIN(yesno ${YESNO_PLUGIN_SOURCES}
    STORAGE_ENGINE
    MODULE_ONLY
    LINK_LIBRARIES ${YESNO_PLUGIN_LIBRARIES}
    )
ENDIF()

IF(TARGET yesno)
  IF(TARGET yesno_c_build)
    ADD_DEPENDENCIES(yesno yesno_c_build)
  ENDIF()
  TARGET_LINK_OPTIONS(
    yesno PRIVATE "$<$<PLATFORM_ID:Linux>:LINKER:--exclude-libs,libyesno_c.a>")
  TARGET_INCLUDE_DIRECTORIES(
    yesno PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${YESNO_C_INCLUDE_DIR}")
  IF(YESNO_WITH_FLIGHT)
    TARGET_COMPILE_DEFINITIONS(yesno PRIVATE YESNO_WITH_FLIGHT=1)
    IF(TARGET Arrow::arrow_shared AND TARGET ArrowFlight::arrow_flight_shared)
      GET_TARGET_PROPERTY(YESNO_ARROW_SONAME Arrow::arrow_shared
                          IMPORTED_SONAME_RELEASE)
      GET_TARGET_PROPERTY(YESNO_ARROW_FLIGHT_SONAME
                          ArrowFlight::arrow_flight_shared
                          IMPORTED_SONAME_RELEASE)
      GET_TARGET_PROPERTY(YESNO_ARROW_LOCATION Arrow::arrow_shared
                          IMPORTED_LOCATION_RELEASE)
      GET_TARGET_PROPERTY(YESNO_ARROW_FLIGHT_LOCATION
                          ArrowFlight::arrow_flight_shared
                          IMPORTED_LOCATION_RELEASE)
      GET_FILENAME_COMPONENT(YESNO_ARROW_LIBRARY_DIR
                             "${YESNO_ARROW_LOCATION}" DIRECTORY)
      GET_FILENAME_COMPONENT(YESNO_ARROW_FLIGHT_LIBRARY_DIR
                             "${YESNO_ARROW_FLIGHT_LOCATION}" DIRECTORY)
      SET(YESNO_ARROW_LINKER_FILE
          "${YESNO_ARROW_LIBRARY_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}arrow${CMAKE_SHARED_LIBRARY_SUFFIX}")
      SET(YESNO_ARROW_FLIGHT_LINKER_FILE
          "${YESNO_ARROW_FLIGHT_LIBRARY_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}arrow_flight${CMAKE_SHARED_LIBRARY_SUFFIX}")
      IF(NOT EXISTS "${YESNO_ARROW_LINKER_FILE}" OR
         NOT EXISTS "${YESNO_ARROW_FLIGHT_LINKER_FILE}")
        MESSAGE(FATAL_ERROR "Arrow package has no real shared linker files")
      ENDIF()
      INSTALL(CODE
        "FILE(MAKE_DIRECTORY \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/lib/private\")
         FILE(COPY_FILE
              \"${YESNO_ARROW_LINKER_FILE}\"
              \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/lib/private/${YESNO_ARROW_SONAME}\"
              ONLY_IF_DIFFERENT)
         FILE(COPY_FILE
              \"${YESNO_ARROW_FLIGHT_LINKER_FILE}\"
              \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/lib/private/${YESNO_ARROW_FLIGHT_SONAME}\"
              ONLY_IF_DIFFERENT)"
        COMPONENT SharedLibraries)
    ENDIF()
  ENDIF()
ENDIF()
